// Scratch differential program (not committed in Prysm): runs the real generated
// v1alpha1.BeaconBlockHeader.UnmarshalSSZ and the hand-written primitives.ValidatorIndex.UnmarshalSSZ on the
// cases of fv/kraken/gen/checks/beacon_block_header_differential.py and prints the observed results as JSON. Expected values are computed
// by that script independently (Python struct.pack), never from this program.
package main

import (
	"encoding/hex"
	"encoding/json"
	"errors"
	"os"

	"github.com/OffchainLabs/methodical-ssz/ssz"
	"github.com/OffchainLabs/prysm/v7/consensus-types/primitives"
	ethpb "github.com/OffchainLabs/prysm/v7/proto/prysm/v1alpha1"
)

type Case struct {
	ID string `json:"id"`
	Op string `json:"op"`
	In string `json:"in"`
}

type Res struct {
	ID      string `json:"id"`
	Err     string `json:"err"`
	ErrSize bool   `json:"err_is_ErrSize"`
	Slot    uint64 `json:"slot"`
	Prop    uint64 `json:"proposer"`
	Roots   []string `json:"roots"`
	Lens    []int  `json:"lens"`
	Caps    []int  `json:"caps"`
	VI      uint64 `json:"vi"`
}

func main() {
	var cs []Case
	if err := json.NewDecoder(os.Stdin).Decode(&cs); err != nil {
		panic(err)
	}
	out := []Res{}
	for _, c := range cs {
		in, err := hex.DecodeString(c.In)
		if err != nil {
			panic(err)
		}
		r := Res{ID: c.ID}
		switch c.Op {
		case "BeaconBlockHeader.UnmarshalSSZ":
			h := &ethpb.BeaconBlockHeader{}
			e := h.UnmarshalSSZ(in)
			if e != nil {
				r.Err = e.Error()
				r.ErrSize = errors.Is(e, ssz.ErrSize)
			}
			r.Slot, r.Prop = uint64(h.Slot), uint64(h.ProposerIndex)
			for _, b := range [][]byte{h.ParentRoot, h.StateRoot, h.BodyRoot} {
				r.Roots = append(r.Roots, hex.EncodeToString(b))
				r.Lens = append(r.Lens, len(b))
				r.Caps = append(r.Caps, cap(b))
			}
		case "ValidatorIndex.UnmarshalSSZ":
			var v primitives.ValidatorIndex
			e := v.UnmarshalSSZ(in)
			if e != nil {
				r.Err = e.Error()
			}
			r.VI = uint64(v)
		default:
			panic(c.Op)
		}
		out = append(out, r)
	}
	json.NewEncoder(os.Stdout).Encode(out)
}

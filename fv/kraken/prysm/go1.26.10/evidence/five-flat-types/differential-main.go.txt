// Scratch differential program (not committed in Prysm): runs the real generated
// v1alpha1.{HistoricalSummary,SigningData,SyncAggregatorSelectionData,ValidatorBalance,VoluntaryExit}.UnmarshalSSZ
// on the cases of fv/kraken/gen/checks/five_flat_types_differential.py and prints the observed results as JSON. Expected values are
// computed by that script independently (Python struct.pack), never from this program. (Text of ../pending-deposit-and-pow-block/differential-main.go.txt.)
package main

import (
	"encoding/hex"
	"encoding/json"
	"errors"
	"os"

	"github.com/OffchainLabs/methodical-ssz/ssz"
	ethpb "github.com/OffchainLabs/prysm/v7/proto/prysm/v1alpha1"
)

type Case struct {
	ID string `json:"id"`
	Op string `json:"op"`
	In string `json:"in"`
}

type Res struct {
	ID      string   `json:"id"`
	Err     string   `json:"err"`
	ErrSize bool     `json:"err_is_ErrSize"`
	U64s    []uint64 `json:"u64s"`
	Bytes   []string `json:"bytes"`
	Lens    []int    `json:"lens"`
	Caps    []int    `json:"caps"`
}

func put(r *Res, bs ...[]byte) {
	for _, b := range bs {
		r.Bytes = append(r.Bytes, hex.EncodeToString(b))
		r.Lens = append(r.Lens, len(b))
		r.Caps = append(r.Caps, cap(b))
	}
}

func main() {
	var cs []Case
	if err := json.NewDecoder(os.Stdin).Decode(&cs); err != nil {
		panic(err)
	}
	out := []Res{}
	for _, c := range cs {
		in, err := hex.DecodeString(c.In)
		if err != nil {
			panic(err)
		}
		r := Res{ID: c.ID}
		var e error
		switch c.Op {
		case "HistoricalSummary.UnmarshalSSZ":
			h := &ethpb.HistoricalSummary{}
			e = h.UnmarshalSSZ(in)
			put(&r, h.BlockSummaryRoot, h.StateSummaryRoot)
		case "SigningData.UnmarshalSSZ":
			h := &ethpb.SigningData{}
			e = h.UnmarshalSSZ(in)
			put(&r, h.ObjectRoot, h.Domain)
		case "SyncAggregatorSelectionData.UnmarshalSSZ":
			h := &ethpb.SyncAggregatorSelectionData{}
			e = h.UnmarshalSSZ(in)
			r.U64s = []uint64{uint64(h.Slot), h.SubcommitteeIndex}
		case "ValidatorBalance.UnmarshalSSZ":
			h := &ethpb.ValidatorBalance{}
			e = h.UnmarshalSSZ(in)
			r.U64s = []uint64{uint64(h.Index), h.Balance}
		case "VoluntaryExit.UnmarshalSSZ":
			h := &ethpb.VoluntaryExit{}
			e = h.UnmarshalSSZ(in)
			r.U64s = []uint64{uint64(h.Epoch), uint64(h.ValidatorIndex)}
		default:
			panic(c.Op)
		}
		if e != nil {
			r.Err = e.Error()
			r.ErrSize = errors.Is(e, ssz.ErrSize)
		}
		out = append(out, r)
	}
	json.NewEncoder(os.Stdout).Encode(out)
}
